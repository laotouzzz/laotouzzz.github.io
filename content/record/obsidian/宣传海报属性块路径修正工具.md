# 一、整体核心定位
这是**带可视化拖拽、支持单文件+文件夹批量处理**的 Obsidian 笔记专用 Python 小工具，专门针对**电影/影视类 Markdown 笔记**，自动同步正文海报链接到顶部 YAML 属性，**路径不一致自动强制修正**，全程无需手动改笔记内容。
# 二、支持的操作方式
1. **拖拽单个 `.md` 笔记文件**：单独处理单篇笔记
2. **拖拽整个文件夹**：**递归批量**遍历文件夹内**所有子目录 + 所有 .md 文件**全自动处理
3. 纯图形界面，不用改代码、不用输路径，拖进去就运行
# 三、核心自动处理逻辑（全覆盖所有场景）
### 1. 自动识别正文海报路径
自动匹配笔记正文固定格式：
```markdown
![宣传海报](00-assets/movie/伸冤人.webp)
```
自动提取括号里的真实海报路径，作为**标准正确路径**。
### 2. YAML 属性智能判断 & 自动修复
#### 场景①：YAML 已有「宣传海报」字段，但路径不一致
- 自动对比 YAML 里的路径 和 正文真实海报路径
- **不匹配直接自动修改 YAML 路径**，保持两边完全一致
#### 场景②：YAML 存在，但没有「宣传海报」字段
- 自动在现有 YAML 末尾**追加一行**：
  ```yaml
  宣传海报: 00-assets/movie/伸冤人.webp
  ```
#### 场景③：笔记顶部完全没有 YAML 属性块
- 自动在笔记最开头**新建完整 YAML 块**：
  ```yaml
  ---
  宣传海报: 00-assets/movie/伸冤人.webp
  ---
  ```
#### 场景④：路径已经完全一致
- 直接跳过，不做任何修改，无重复写入
## 四、智能容错与提示功能
1. 自动过滤**非 .md 文件**，不会乱处理其他格式
2. 笔记正文找不到 `![宣传海报]()` 时，自动跳过并提示
3. 文件编码统一用 **UTF-8**，不会出现中文乱码
4. 单个文件处理：弹窗显示精简结果
5. 文件夹批量处理：汇总所有文件处理状态，超过20条自动省略并统计总数
6. 捕获异常报错，文件损坏/权限不足都会友好提示，不闪退
## 七、运行依赖
仅需安装一次依赖库：
```bash
pip install tkinterdnd2
```
装好后直接双击脚本即可运行，无需任何配置。
# 主要python代码
```
import re
import os
import tkinter as tk
from tkinter import messagebox
from pathlib import Path
from tkinterdnd2 import TkinterDnD, DND_FILES

# ====================== 核心处理函数 ======================
def process_markdown_file(file_path: str | Path):
    try:
        file = Path(file_path)
        if not file.exists() or file.suffix != ".md":
            return f"❌ 跳过 {file.name}：不是有效的 .md 文件"

        content = file.read_text(encoding="utf-8")

        # 1. 提取正文中的宣传海报路径
        poster_match = re.search(r"!\[宣传海报\]\((.+?)\)", content)
        if not poster_match:
            return f"⚠️ 跳过 {file.name}：未找到宣传海报"
        real_poster = poster_match.group(1).strip()

        # 2. 匹配 YAML 块
        yaml_pattern = re.compile(r"^---\n(.*?)\n---", re.DOTALL | re.MULTILINE)
        yaml_match = yaml_pattern.search(content)

        if yaml_match:
            yaml_content = yaml_match.group(1)
            if "宣传海报:" in yaml_content:
                # 已存在 → 检查是否匹配
                yaml_poster_match = re.search(r"宣传海报:\s*(.+)", yaml_content)
                if yaml_poster_match:
                    yaml_poster = yaml_poster_match.group(1).strip()
                    if yaml_poster == real_poster:
                        return f"✅ {file.name}：已匹配，无需修改"
                    else:
                        # 不匹配 → 自动修正
                        new_yaml = re.sub(r"宣传海报:.+", f"宣传海报: {real_poster}", yaml_content)
                        new_content = content.replace(yaml_match.group(0), f"---\n{new_yaml}\n---")
                        file.write_text(new_content, encoding="utf-8")
                        return f"🔧 {file.name}：已自动修正海报路径"
                else:
                    return f"⚠️ {file.name}：YAML 格式错误"
            else:
                # 无字段 → 添加
                new_yaml = yaml_content.rstrip() + f"\n宣传海报: {real_poster}\n"
                new_content = content.replace(yaml_match.group(0), f"---\n{new_yaml}---")
                file.write_text(new_content, encoding="utf-8")
                return f"✅ {file.name}：已添加宣传海报字段"
        else:
            # 无 YAML → 创建
            new_content = f"---\n宣传海报: {real_poster}\n---\n{content}"
            file.write_text(new_content, encoding="utf-8")
            return f"✅ {file.name}：已创建 YAML 并添加海报"

    except Exception as e:
        return f"❌ {file.name}：处理失败 - {str(e)}"

# ====================== 批量处理 ======================
def process_folder(folder_path: str | Path):
    folder = Path(folder_path)
    if not folder.is_dir():
        return ["❌ 不是有效文件夹"]

    results = []
    md_files = list(folder.rglob("*.md"))
    if not md_files:
        return ["⚠️ 文件夹内没有找到 .md 文件"]

    for md in md_files:
        res = process_markdown_file(md)
        results.append(res)

    return results

# ====================== 拖拽处理 ======================
def on_drop(event):
    raw_path = event.data.strip("{}")
    path = Path(raw_path)

    if path.is_file():
        # 单个文件
        result = process_markdown_file(path)
        messagebox.showinfo("处理完成", result)
    elif path.is_dir():
        # 批量文件夹
        results = process_folder(path)
        log = "\n".join(results[:20])  # 最多显示20条
        if len(results) > 20:
            log += f"\n...（共{len(results)}条结果）"
        messagebox.showinfo("批量处理完成", log)
    else:
        messagebox.showerror("错误", "无效路径")

# ====================== 界面 ======================
root = TkinterDnD.Tk()
root.title("Obsidian 宣传海报 YAML 自动修正工具 - 批量版")
root.geometry("560x320")
root.resizable(False, False)

label = tk.Label(
    root,
    text="📌 拖拽 单个MD文件 或 整个文件夹 到这里\n\n自动检查 | 自动匹配 | 自动修复 | 批量处理",
    font=("Microsoft YaHei", 14),
    bg="#f8f9fa",
    fg="#2d3436"
)
label.pack(expand=True, fill="both", pady=20)

tip = tk.Label(root, text="支持：批量处理所有电影笔记 | 自动修正 YAML 海报路径", font=("Microsoft YaHei", 10), fg="#636e72")
tip.pack(pady=5)

root.drop_target_register(DND_FILES)
root.dnd_bind('<<Drop>>', on_drop)

root.mainloop()
```
